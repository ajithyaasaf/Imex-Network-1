## Advanced React Spring Implementation

### Hero Parallax Effect
```typescript
function HeroSection() {
  const [{ scrollY }, setScrollY] = useSpring(() => ({ scrollY: 0 }));
  
  useEffect(() => {
    const handleScroll = () => setScrollY({ scrollY: window.scrollY });
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  const buildingTransform = scrollY.to(y => `translate3d(0, ${y * 0.5}px, 0)`);
  const overlayTransform = scrollY.to(y => `translate3d(0, ${y * 0.2}px, 0)`);
  const contentOpacity = scrollY.to([0, 300], [1, 0]);

  return (
    <div className="relative h-screen overflow-hidden">
      <animated.div 
        className="absolute inset-0 will-change-transform"
        style={{ transform: buildingTransform }}
      >
        <img 
          src="/images/hero-building.jpg" 
          alt="Luxury Modern Building"
          className="w-full h-full object-cover"
        />
      </animated.div>
      
      <animated.div 
        className="absolute inset-0 bg-gradient-to-r from-red-600/80 to-transparent"
        style={{ 
          transform: overlayTransform,
          opacity: contentOpacity 
        }}
      >
        <div className="container mx-auto h-full flex items-center">
          <div className="text-white max-w-2xl">
            <h1 className="text-6xl font-bold mb-4">
              WE MAKE YOUR PLACE
              <span className="block text-yellow-300">BETTER THAN OTHERS</span>
            </h1>
            <p className="text-xl mb-8">Global Import, Export & Sourcing Partner</p>
            <div className="flex gap-4">
              <Button variant="primary" size="lg">Start Your Project</Button>
              <Button variant="outline" size="lg">View Portfolio</Button>
            </div>
          </div>
        </div>
      </animated.div>
    </div>
  );
}
```

### Process Timeline Animation
```typescript
function ProcessTimeline() {
  const [ref, inView] = useInView({ threshold: 0.3, triggerOnce: true });
  
  const timelineRef = useSpringRef();
  const stepsRef = useSpringRef();
  
  const roadAnimation = useSpring({
    ref: timelineRef,
    from: { strokeDashoffset: 2000 },
    to: { strokeDashoffset: inView ? 0 : 2000 },
    config: { duration: 2000 }
  });

  const stepsAnimation = useTrail(8, {
    ref: stepsRef,
    opacity: inView ? 1 : 0,
    transform: inView ? 'translateY(0px) scale(1)' : 'translateY(50px) scale(0.8)',
    config: { tension: 280, friction: 60 }
  });

  useChain(inView ? [timelineRef, stepsRef] : [], [0, 0.5]);

  return (
    <section ref={ref} className="py-20 bg-gray-50">
      <div className="container mx-auto">
        <h2 className="text-4xl font-bold text-center mb-16">Our Process</h2>
        
        <div className="relative">
          <svg className="absolute top-1/2 left-0 w-full h-2 -translate-y-1/2">
            <animated.path
              d="M50,50 Q200,20 400,50 T750,50 Q900,80 1200,50"
              stroke="var(--primary-red)"
              strokeWidth="4"
              fill="none"
              strokeDasharray="20 10"
              style={roadAnimation}
            />
          </svg>
          
          <div className="grid grid-cols-4 gap-8 relative z-10">
            {stepsAnimation.map((style, index) => (
              <animated.div key={index} style={style}>
                <ProcessStep 
                  number={index + 1}
                  title={processSteps[index].title}
                  description={processSteps[index].description}
                  icon={processSteps[index].icon}
                />
              </animated.div>
            ))}
          </div>
        </div>
      </div>
    </section>
  );
}
```

```typescript
function PortfolioGallery({ projects }: { projects: Project[] }) {
  const [selectedCategory, setSelectedCategory] = useState<string>('all');
  const [ref, inView] = useInView({ threshold: 0.2 });
  
  const filteredProjects = selectedCategory === 'all' 
    ? projects 
    : projects.filter(p => p.category === selectedCategory);

  const transitions = useTransition(filteredProjects, {
    from: { opacity: 0, transform: 'scale(0.9) translateY(40px)' },
    enter: { opacity: 1, transform: 'scale(1) translateY(0px)' },
    leave: { opacity: 0, transform: 'scale(0.9) translateY(-40px)' },
    trail: 100,
    config: { tension: 300, friction: 30 }
  });

  return (
    <section className="py-20">
      <div className="container mx-auto">
        <div className="text-center mb-12">
          <h2 className="text-4xl font-bold mb-6">Project Portfolio</h2>
          
          <div className="flex justify-center gap-4 mb-8">
            {categories.map(category => (
              <button
                key={category.id}
                onClick={() => setSelectedCategory(category.id)}
                className={`px-6 py-3 rounded-full transition-all ${
                  selectedCategory === category.id
                    ? 'bg-red-600 text-white'
                    : 'bg-gray-200 text-gray-700 hover:bg-gray-300'
                }`}
              >
                {category.name}
              </button>
            ))}
          </div>
        </div>

        <div ref={ref} className="masonry-grid">
          {transitions((style, project) => (
            <animated.div style={style}>
              <ProjectCard 
                project={project}
                onView={() => openProjectModal(project)}
              />
            </animated.div>
          ))}
        </div>
      </div>
    </section>
  );
}
```
